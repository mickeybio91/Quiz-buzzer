  <script>
    // ⚠️ 아래 URL을 본인의 Apps Script 배포 URL로 변경하세요.
    const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbx...YOUR_DEPLOYMENT_ID.../exec";

    let myName = "";
    let isTeacher = false;
    let timer = null;
    let currentInterval = 2500;
    let hasBuzzedCurrentQuiz = false;

    // GitHub Pages용 공통 API 통신 함수
    function callApi(action, data = {}) {
      const payload = { action: action, data: data };
      return fetch(APPS_SCRIPT_URL, {
        method: 'POST',
        mode: 'cors',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify(payload)
      })
      .then(res => res.json())
      .catch(err => {
        console.error("API Error:", err);
      });
    }

    function login() {
      myName = document.getElementById('nickname').value.trim();
      const pin = document.getElementById('pin').value;
      if (!myName) return alert('닉네임을 입력하세요.');

      if (pin === "1234") isTeacher = true;

      document.getElementById('login-sec').style.display = 'none';

      // 유저 등록 및 대기실 상태 가져오기
      callApi('joinUser', { nickname: myName }).then(state => {
        if (state) render(state);
      });
      startPolling(800);
    }

    function startPolling(intervalMs) {
      if (timer) clearInterval(timer);
      currentInterval = intervalMs;
      timer = setInterval(fetchState, intervalMs);
    }

    function fetchState() {
      callApi('getState').then(state => {
        if (state) render(state);
      });
    }

    function render(state) {
      if (!state) return;

      // 1. 수업 종료 처리
      if (state.status === 'END') {
        if (timer) clearInterval(timer);
        showReport(state.history);
        return;
      }

      // 2. 대기실(LOBBY) 상태 렌더링
      if (state.status === 'LOBBY') {
        document.getElementById('lobby-sec').style.display = 'block';
        document.getElementById('app-sec').style.display = 'none';
        document.getElementById('report-sec').style.display = 'none';

        if (isTeacher) {
          document.getElementById('teacher-start-ctrl').style.display = 'block';
        }

        const users = state.users || [];
        document.getElementById('user-count').innerText = users.length;
        const userListDiv = document.getElementById('user-list');
        userListDiv.innerHTML = '';
        users.forEach(u => {
          userListDiv.innerHTML += `<span class="user-tag">👤 ${u}</span>`;
        });
        return;
      }

      // 3. 수업 진행(WAIT / OPEN) 상태 렌더링
      document.getElementById('lobby-sec').style.display = 'none';
      document.getElementById('app-sec').style.display = 'block';
      document.getElementById('report-sec').style.display = 'none';

      if (isTeacher) {
        document.getElementById('teacher-ctrl').style.display = 'block';
        document.getElementById('th-ctrl').style.display = 'table-cell';
      } else {
        document.getElementById('student-ctrl').style.display = 'block';
      }

      document.getElementById('quiz-title').innerText = `문제 ${state.quizNum}`;
      const banner = document.getElementById('status-banner');
      const buzzBtn = document.getElementById('buzz-btn');

      if (state.status === 'OPEN') {
        banner.className = "status-banner status-active";
        banner.innerHTML = "🔔 부저 오픈! 지금 누르세요!!";
        
        if (!isTeacher && !hasBuzzedCurrentQuiz) {
          buzzBtn.disabled = false;
        }
        if (currentInterval !== 500) startPolling(500);
      } else {
        banner.className = "status-banner status-wait";
        banner.innerHTML = "🔒 부저 대기 중입니다";
        
        if (!isTeacher) buzzBtn.disabled = true;
        hasBuzzedCurrentQuiz = false;
        
        if (currentInterval !== 1000) startPolling(1000);
      }

      const tbody = document.getElementById('buzz-list');
      tbody.innerHTML = '';
      const buzzes = state.buzzes || [];
      buzzes.forEach((b, idx) => {
        let actionBtns = '';
        if (isTeacher) {
          actionBtns = `<button class="btn-pass" onclick="judge(${idx}, 'PASS')">PASS</button>
                        <button class="btn-fail" onclick="judge(${idx}, 'FAIL')">FAIL</button>`;
        }
        
        const isTop1 = (idx === 0) ? 'class="top1-row"' : '';
        const rankText = (idx === 0) ? '🏆 1등' : `${idx + 1}`;

        let rowHtml = `<tr ${isTop1}>
          <td>${rankText}</td>
          <td>${b.nickname}</td>
          <td>${b.result}</td>`;
        if (isTeacher) rowHtml += `<td>${actionBtns}</td>`;
        rowHtml += `</tr>`;

        tbody.innerHTML += rowHtml;
      });
    }

    function startLesson() {
      callApi('startLesson').then(render);
    }

    function openBuzzer() { 
      const banner = document.getElementById('status-banner');
      banner.className = "status-banner status-active";
      banner.innerHTML = "🔔 부저 오픈! (전송 중...)";
      callApi('openBuzzer').then(render); 
    }

    function judge(idx, res) { 
      const tbody = document.getElementById('buzz-list');
      if (tbody && tbody.rows[idx]) {
        tbody.rows[idx].cells[2].innerText = res; 
      }
      callApi('judgeStudent', { index: idx, result: res }).then(render); 
    }

    function nextQuiz() { 
      const banner = document.getElementById('status-banner');
      banner.className = "status-banner status-wait";
      banner.innerHTML = "🔒 다음 문제로 이동 중...";
      callApi('nextQuiz').then(render); 
    }

    function endLesson() { 
      if (confirm('수업을 종료하고 최종 결과를 보시겠습니까?')) {
        callApi('endLesson').then(render); 
      }
    }

    function resetAll() {
      if (confirm('모든 기록과 접속자 명단을 초기화하시겠습니까?')) {
        callApi('resetSystem').then(res => {
          render(res);
          startPolling(800);
        });
      }
    }

    function pressBuzzer() {
      const buzzBtn = document.getElementById('buzz-btn');
      buzzBtn.disabled = true;
      hasBuzzedCurrentQuiz = true;

      callApi('pressBuzzer', { nickname: myName }).then(res => {
        if (res && !res.success && res.msg) alert(res.msg);
        fetchState();
      });
    }

    function showReport(history = {}) {
      document.getElementById('lobby-sec').style.display = 'none';
      document.getElementById('app-sec').style.display = 'none';
      document.getElementById('report-sec').style.display = 'block';
      
      let html = "";

      if (isTeacher) {
        document.getElementById('teacher-report-ctrl').style.display = 'block';
        for (let q in history) {
          html += `<h3 style="margin-top:20px; text-align:left;">[문제 ${q}]</h3>
          <table><tr><th>순위</th><th>닉네임</th><th>결과</th></tr>`;
          if (history[q].length === 0) {
            html += `<tr><td colspan="3">참가자 없음</td></tr>`;
          } else {
            history[q].forEach((b, idx) => {
              html += `<tr><td>${idx + 1}</td><td>${b.nickname}</td><td>${b.result}</td></tr>`;
            });
          }
          html += `</table>`;
        }
      } else {
        document.getElementById('teacher-report-ctrl').style.display = 'none';
        html += `<p style="font-size:18px; font-weight:bold; color:#0369a1;">👤 ${myName} 학생의 참여 기록</p>`;
        html += `<table><tr><th>문제 번호</th><th>내 순위</th><th>최종 판정</th></tr>`;
        
        let participated = false;
        for (let q in history) {
          const myIndex = history[q].findIndex(b => b.nickname === myName);
          if (myIndex !== -1) {
            participated = true;
            const myRecord = history[q][myIndex];
            html += `<tr>
              <td>문제 ${q}</td>
              <td>${myIndex + 1}등</td>
              <td>${myRecord.result}</td>
            </tr>`;
          }
        }
        if (!participated) {
          html += `<tr><td colspan="3">부저를 누른 기록이 없습니다.</td></tr>`;
        }
        html += `</table>`;
      }

      document.getElementById('report-content').innerHTML = html;
    }

    function downloadPDF() {
      const element = document.getElementById('report-content');
      html2pdf().from(element).save('수업_발표_결과.pdf');
    }
  </script>
